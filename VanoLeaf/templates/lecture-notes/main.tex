\begin{definition}{Group}{group}
A group is a set $G$ with an associative operation, a unit and inverses.
\end{definition}
\begin{theorem}{Lagrange}{lagrange}
If $H\le G$ and $G$ is finite, then $|H|$ divides $|G|$.
\end{theorem}
\begin{proof}
The cosets of $H$ partition $G$ and all have $|H|$ elements.
\end{proof}
